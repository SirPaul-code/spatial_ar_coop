cmake_minimum_required(VERSION 3.20)
project(StableAR VERSION 0.2.0 LANGUAGES C CXX)
option(STABLEAR_BUILD_SHARED "Build shared library" OFF)
option(STABLEAR_BUILD_TESTS "Build tests" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(_kind STATIC)
if(STABLEAR_BUILD_SHARED)
  set(_kind SHARED)
endif()
add_library(stablear ${_kind}
  src/geometry.cpp src/surface.cpp src/frame_ledger.cpp src/attachment.cpp
  src/session.cpp src/entitlement.cpp src/stablear_c.cpp)
add_library(StableAR::stablear ALIAS stablear)
target_include_directories(stablear PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(stablear PUBLIC cxx_std_20)
if(STABLEAR_BUILD_SHARED)
  target_compile_definitions(stablear PRIVATE STABLEAR_BUILDING_DLL)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS stablear EXPORT StableARTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/StableARConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/StableARConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StableAR)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/StableARConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT StableARTargets FILE StableARTargets.cmake NAMESPACE StableAR:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StableAR)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/StableARConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/StableARConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/StableAR)
if(STABLEAR_BUILD_TESTS)
 enable_testing()
 add_executable(stablear_contract tests/contract.cpp)
 target_link_libraries(stablear_contract PRIVATE stablear)
 add_test(NAME stablear_contract COMMAND stablear_contract)
 add_executable(stablear_c_smoke tests/c_smoke.c)
 set_target_properties(stablear_c_smoke PROPERTIES LINKER_LANGUAGE CXX)
 target_link_libraries(stablear_c_smoke PRIVATE stablear)
 add_test(NAME stablear_c_smoke COMMAND stablear_c_smoke)
endif()
