cmake_minimum_required(VERSION 3.20)
project(StableAROpenXR VERSION 0.2.0 LANGUAGES CXX)
option(STABLEAR_OPENXR_BUILD_TESTS "Build OpenXR coordinate contract" ON)
if(NOT TARGET StableAR::stablear)
  set(STABLEAR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../native stablear-native EXCLUDE_FROM_ALL)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENXR REQUIRED IMPORTED_TARGET openxr)
add_library(stablear_openxr STATIC src/openxr_adapter.cpp)
add_library(StableAR::openxr ALIAS stablear_openxr)
target_include_directories(stablear_openxr PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(stablear_openxr PUBLIC StableAR::stablear PkgConfig::OPENXR)
target_compile_features(stablear_openxr PUBLIC cxx_std_20)
if(STABLEAR_OPENXR_BUILD_TESTS)
  enable_testing()
  add_executable(stablear_openxr_contract tests/coordinate_contract.cpp)
  target_link_libraries(stablear_openxr_contract PRIVATE stablear_openxr)
  add_test(NAME stablear_openxr_contract COMMAND stablear_openxr_contract)
endif()
